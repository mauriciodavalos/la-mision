--   2) Revísalo contra el modelo de auth definitivo (roles, claims).
--   3) Aplícalo.
--
-- La regla de oro: un agente solo ve datos de los clientes a los que pertenece
-- (tabla agente_cliente), resuelto contra auth.uid() -> agentes.user_id.
--
-- OJO (desde 0006_asignaciones.sql): la membresía ya no es el único filtro.
--  * `agente_asignacion` define QUÉ marca y en QUÉ cadena captura cada agente.
--    Estas policies dan alcance a nivel CLIENTE; afinarlas a marca/cadena es el
--    siguiente escalón y debe hacerse aquí, no solo en la app.
--  * `agentes.es_admin` se salta todo. Ojo al escribirlo: un admin tiene que
--    poder leer y escribir en cualquier cliente.

-- Helper: ¿el usuario autenticado es administrador? --------------------------
create or replace function public.es_admin()
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select coalesce(
    (select a.es_admin from public.agentes a where a.user_id = auth.uid()),
    false
  );
$$;

-- Helper: clientes a los que pertenece el usuario autenticado -----------------
-- Un admin los ve todos.
create or replace function public.clientes_del_usuario()
returns setof uuid
language sql
stable
security definer
set search_path = public
as $$
  select c.id from public.clientes c where public.es_admin()
  union
  select ac.cliente_id
  from public.agente_cliente ac
  join public.agentes a on a.id = ac.agente_id
  where a.user_id = auth.uid();
$$;

-- Activar RLS ----------------------------------------------------------------
alter table public.clientes       enable row level security;
alter table public.marcas         enable row level security;
alter table public.cadenas        enable row level security;
alter table public.tiendas        enable row level security;
alter table public.agentes        enable row level security;
alter table public.agente_cliente enable row level security;
alter table public.visitas        enable row level security;
alter table public.evidencias     enable row level security;
alter table public.agente_asignacion enable row level security;

-- El agente ve sus propias asignaciones (la app las necesita para filtrar).
create policy asignacion_propia on public.agente_asignacion
  for select using (
    public.es_admin()
    or agente_id in (select id from public.agentes where user_id = auth.uid())
  );

-- Policies por cliente_id (lectura/escritura del propio cliente) --------------
-- Patrón repetido para cada tabla operativa con cliente_id:
create policy cliente_scope_marcas on public.marcas
  for all using (cliente_id in (select public.clientes_del_usuario()))
  with check (cliente_id in (select public.clientes_del_usuario()));

create policy cliente_scope_cadenas on public.cadenas
  for all using (cliente_id in (select public.clientes_del_usuario()))
  with check (cliente_id in (select public.clientes_del_usuario()));

create policy cliente_scope_tiendas on public.tiendas
  for all using (cliente_id in (select public.clientes_del_usuario()))
  with check (cliente_id in (select public.clientes_del_usuario()));

create policy cliente_scope_visitas on public.visitas
  for all using (cliente_id in (select public.clientes_del_usuario()))
  with check (cliente_id in (select public.clientes_del_usuario()));

create policy cliente_scope_evidencias on public.evidencias
  for all using (cliente_id in (select public.clientes_del_usuario()))
  with check (cliente_id in (select public.clientes_del_usuario()));

-- clientes: el usuario solo ve los clientes a los que pertenece
create policy cliente_scope_clientes on public.clientes
  for select using (id in (select public.clientes_del_usuario()));

-- agentes / agente_cliente: el usuario ve su propia ficha y membresías.
-- OJO: la pantalla de identificación necesita listar a los agentes de un cliente
-- ANTES de que nadie se autentique. Con RLS puesta eso deja de funcionar, así que
-- al aplicar esta migración hay que sustituir el PIN por Supabase Auth de verdad
-- (que es justo el punto de la fase 2), no dejar las dos cosas conviviendo.
create policy agente_propio on public.agentes
  for select using (public.es_admin() or user_id = auth.uid());

create policy agente_cliente_propio on public.agente_cliente
  for select using (
    agente_id in (select id from public.agentes where user_id = auth.uid())
  );

-- Storage (bucket 'evidencias'): acceso solo a carpetas de tus clientes.
-- La ruta es evidencias/{slug_cliente}/{slug_cadena}/{clave-tienda}/{archivo},
-- así que el PRIMER segmento debe ser el slug de un cliente del usuario.
--
-- OJO: antes de 0005_slugs.sql el primer segmento era el cliente_id (UUID). Si
-- quedaron objetos viejos con ruta por UUID, hay que migrarlos ANTES de aplicar
-- esta policy, o quedarán inaccesibles. Para revisarlo:
--   select name from storage.objects
--    where bucket_id = 'evidencias'
--      and (storage.foldername(name))[1] ~ '^[0-9a-f-]{36}$';
create policy evidencias_storage_scope on storage.objects
  for all using (
    bucket_id = 'evidencias'
    and (storage.foldername(name))[1] in (
      select c.slug from public.clientes c
      where c.id in (select public.clientes_del_usuario())
    )
  )
  with check (
    bucket_id = 'evidencias'
    and (storage.foldername(name))[1] in (
      select c.slug from public.clientes c
      where c.id in (select public.clientes_del_usuario())
    )
  );
